material {
	name : %NAME%,
	requires : [ ],
	parameters : [
		{
			type : float4,
			name : base_color
		}
	],
	shadingModel : unlit,
	flipUV : false,
	blending : transparent
}

fragment {
	void material(inout MaterialInputs material) {
		prepareMaterial(material);
		
#option ShadingMode
#variant Plain
		material.baseColor.rgba = materialParams.base_color.rgba;
#variant Plain3D
		material.baseColor.rgb = materialParams.base_color.rgb;
		material.baseColor.a = materialParams.base_color.a;
#endoption

	}
}
