material {
	name : %NAME%,
	requires : [ tangents ],
	parameters : [
		{
			type : float4,
			name : base_color
		}
	],
	shadingModel : unlit,
	flipUV : false,
	blending : transparent
}

fragment {
	void material(inout MaterialInputs material) {
		prepareMaterial(material);
		
#option ShadingMode
#variant Plain
		material.baseColor.rgba = materialParams.base_color.rgba;
#variant Plain3D
        float dot = clamp(dot(getWorldGeometricNormalVector(), getWorldViewVector()), 0.0, 1.0);
		material.baseColor.rgb = materialParams.base_color.rgb * (0.25 + (0.75 * (dot * dot)));
		material.baseColor.a = materialParams.base_color.a;
#endoption

	}
}
