material {
	name : %NAME%,
#option ShadingMode
#variant Plain
	requires : [ tangents ],
	blending : transparent,
#variant Plain3D
	requires : [ tangents ],
	blending : transparent,
	transparency : twoPassesTwoSides,
#variant Wireframe
	requires : [ tangents, color ],
	blending : masked,
	culling : none,
#endoption
	parameters : [
		{
			type : float4,
			name : base_color
		}
	],
	shadingModel : unlit,
	flipUV : false,
}

fragment {
	void material(inout MaterialInputs material) {
		prepareMaterial(material);

#option ShadingMode
#variant Plain
		material.baseColor.rgba = materialParams.base_color.rgba;
#variant Plain3D
        float dot = saturate(abs(dot(getWorldGeometricNormalVector(), getWorldViewVector())));
		material.baseColor.rgb = materialParams.base_color.rgb * (0.25 + (0.75 * sq(dot)));
		material.baseColor.a = materialParams.base_color.a;
#variant Wireframe
		vec3 bary = getColor().rgb;
		vec3 baryWidth = fwidth(bary);
		vec3 baryFalloff = smoothstep(vec3(0.0), baryWidth * 2.5, bary);
		float edgeFactor = 1.0 - min(min(baryFalloff.x, baryFalloff.y), baryFalloff.z);
		material.baseColor.rgb = materialParams.base_color.rgb;
		material.baseColor.a = edgeFactor;
#endoption

	}
}
