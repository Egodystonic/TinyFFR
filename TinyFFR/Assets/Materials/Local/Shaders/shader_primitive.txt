material {
	name : %NAME%,
	requires : [ ],
	parameters : [

			type : float4,
			name : base_color
		}
	],
	shadingModel : unlit,
	flipUV : false,
	blending : transparent
}

	void material(inout MaterialInputs material) {

		material.postLightingColor.rgb = materialParams_base_color.rgb;

	}
}
