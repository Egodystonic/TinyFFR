material {
	name : %NAME%,
	parameters : [
		{
			type : sampler2d,
			name : color_map
		}
	],
	requires : [
		uv0,
		color
	],
	shadingModel : unlit,
	flipUV : false,
	culling : none,
	depthCulling : false,
	blending : transparent
}

fragment {
	void material(inout MaterialInputs material) {
		prepareMaterial(material);

		vec4 albedo = texture(materialParams_color_map, getUV0()).rgba;
		material.baseColor = getColor() * albedo;
		material.baseColor.rgb *= material.baseColor.a;
	}
}
