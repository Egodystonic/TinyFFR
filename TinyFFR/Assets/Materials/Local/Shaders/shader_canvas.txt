material {
	name : %NAME%,
	parameters : [
		{
			type : float4x4,
			name : uv_transform
		},
		{
			type : sampler2d,
			name : color_map
		},
		{
			type : sampler2d,
			name : color_map_blend
		},
		{
			type : float,
			name : color_map_blend_distance
		},
		{
			type : float,
			name : opacity
		}
	],
	requires : [
		uv0
	],
	shadingModel : lit,
	flipUV : false,
	postLightingBlending : opaque,
	blending : fade
}

fragment {
	void material(inout MaterialInputs material) {
		float3 uv0Extended = float3(getUV0().xy, 1.0);
		vec2 uv0 = mulMat3x3Float3(materialParams.uv_transform, uv0Extended).xy;

		prepareMaterial(material);

		vec4 startColor = texture(materialParams_color_map, uv0).rgba;
		vec4 endColor = texture(materialParams_color_map_blend, uv0).rgba;
		vec4 blendedColor = startColor * (1.0f - materialParams.color_map_blend_distance) + endColor * materialParams.color_map_blend_distance;
		material.postLightingColor = blendedColor * materialParams.opacity;
	}
}
